# What are the proteins where the APOC1 Homo sapiens gene encodes?
BASE <http://rdf.biogateway.eu/graph/>
PREFIX rdfs: <http://www.w3.org/2000/01/rdf-schema#>
SELECT DISTINCT ?o
WHERE {
   ?s ?r "APOC1" .
   ?s <http://semanticscience.org/resource/SIO_010078> ?o .
   ?s <http://purl.obolibrary.org/obo/RO_0002162> ?t .
   ?t rdfs:label "Homo sapiens" .
}
