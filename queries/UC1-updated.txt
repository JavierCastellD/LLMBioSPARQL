#1. Is the mutation located in any enhancer sequence of the network? Returns the label of the sequences and their coordinates.
# What databases support the sequence? Indicate its label. Is the enhancer related to any disease?

#3. Multiple enhancers can collaborate in the regulation of one gene. What other enhancers can regulate the gene TOX3 in the breast tissue (UBERON_0000310)? We follow three different criteria:
#3.1. Enhancers provided by databases:
#(the gene is in the strand "-", so the TSS of the gene is the end position)
# Returns the enhancer label and its coordinates, as well as the middle position of the sequence, the TSS of the gene, and the distance between the middle position and the TSS.
# Order the results by distance.

PREFIX obo: <http://purl.obolibrary.org/obo/>
PREFIX human_gene: <http://rdf.biogateway.eu/gene/9606/>
PREFIX rdfs: <http://www.w3.org/2000/01/rdf-schema#>
PREFIX skos: <http://www.w3.org/2004/02/skos/core#>
PREFIX rdf: <http://www.w3.org/1999/02/22-rdf-syntax-ns#>
SELECT DISTINCT ?enh_label ?enh_start ?enh_end ?enh_middle ?TSS_gene ?distance
WHERE {
	GRAPH <http://rdf.biogateway.eu/graph/crm2gene> {
		?enhID obo:RO_0002429 human_gene:TOX3 .
	}

    GRAPH <http://rdf.biogateway.eu/graph/gene> {
        human_gene:TOX3 obo:GENO_0000895 ?TSS_gene ;
			obo:BFO_0000050 ?chr .
    }

    GRAPH <http://rdf.biogateway.eu/graph/crm> {
        ?enhID rdfs:subClassOf obo:SO_0000727 ;
            skos:prefLabel ?enh_label ;
			obo:GENO_0000894 ?enh_start ;
			obo:GENO_0000895 ?enh_end ;
			obo:BFO_0000050 ?chr .

		?enh_inst rdf:type ?enhID ;
			obo:TXPO_0003500 obo:UBERON_0000310 .
    }
        
    BIND((?enh_end - ?enh_start) AS ?enh_range)
    BIND((ROUND(?enh_range/2) + ?enh_start) AS ?enh_middle)
    BIND(ABS(?TSS_gene - ?enh_middle) AS ?distance)
}
ORDER BY ?distance

#3.3. Enhancers located in the same TAD as the TOX3 gene, because these domains act as regulatory modules
# Returns the enhancer label and its coordinates, as well as the middle position of the sequence, the TSS of the gene, and the distance between the middle position and the TSS.
# Order the results by distance.

PREFIX obo: <http://purl.obolibrary.org/obo/>
PREFIX human_gene: <http://rdf.biogateway.eu/gene/9606/>
PREFIX rdfs: <http://www.w3.org/2000/01/rdf-schema#>
PREFIX skos: <http://www.w3.org/2004/02/skos/core#>
PREFIX rdf: <http://www.w3.org/1999/02/22-rdf-syntax-ns#>
SELECT DISTINCT ?enh_label ?enh_start ?enh_end ?enh_middle ?TSS_gene ?distance
WHERE {
    GRAPH <http://rdf.biogateway.eu/graph/gene> {
		human_gene:TOX3 obo:GENO_0000895 ?TSS_gene ;
			obo:BFO_0000050 ?chr .
    }

	GRAPH <http://rdf.biogateway.eu/graph/tad> {
		?TAD_ID rdfs:subClassOf obo:SO_0002304 ;
			obo:BFO_0000050 ?chr ;
			obo:GENO_0000894 ?TAD_start ;
			obo:GENO_0000895 ?TAD_end .
		
		?TAD_inst rdf:type ?TAD_ID ;
			obo:TXPO_0003500 obo:UBERON_0000310 .
	}

	GRAPH <http://rdf.biogateway.eu/graph/crm> {
		?enh_ID rdfs:subClassOf obo:SO_0000727 ;
			skos:prefLabel ?enh_label ;
			obo:BFO_0000050 ?chr ;
			obo:GENO_0000894 ?enh_start ;
			obo:GENO_0000895 ?enh_end .

		?enh_inst rdf:type ?enh_ID ;
			obo:TXPO_0003500 obo:UBERON_0000310 .
	}

    BIND (?enh_end - ?enh_start AS ?enh_range)
    BIND (?enh_range/2 AS ?enh_div)
    BIND (?enh_div + ?enh_start AS ?enh_middle)
    BIND (ABS(?TSS_gene - ?enh_middle) AS ?distance)

    FILTER ( ?TAD_start < ?TSS_gene ) .
    FILTER ( ?TAD_end > ?TSS_gene ) .
    FILTER ( ?enh_start > ?TAD_start ) .
    FILTER ( ?enh_end < ?TAD_end ) .
}
ORDER BY ?distance
